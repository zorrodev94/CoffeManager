        internal struct BranchRegistrationData

        }

        /// Contains the branch profile completed after registration. Neighborhood means the local neighborhood or colonia. Null optional values clear their stored value.

        internal struct BranchProfileData
        {

            public string Name { get; set; }

            public string? Description { get; set; }

            [StringLength(100)]
            public string? Neighborhood { get; set; }
            [StringLength(100)]
            public string? City { get; set; }
            [StringLength(100)]
            public string? State { get; set; }
            [StringLength(10)]
            public string? PostalCode { get; set; }
            [StringLength(30)]
            public string? PhoneNumber { get; set; }
            [StringLength(254), EmailAddress]
            public string? Email { get; set; }
        }
        /// <summary>
        /// Contains the branch identity and profile without loading areas, employees or inventory automatically.
        /// </summary>
        internal struct BranchDetailsData
        {
            public Guid BranchId { get; set; }
            public Guid BusinessId { get; set; }
            public DateTimeOffset CreatedAt { get; set; }
            public BranchProfileData Profile { get; set; }
